// ==UserScript==
// @name         Auto Open Details
// @namespace    http://tampermonkey.net/
// @version      1.0
// @match        *://*/*
// @grant        none
// @description  to auto open Details for faster use
// @author       Albert D. Delgado
// ==/UserScript==

(function () {
    'use strict';

    function tryOpenDetails() {
        if (
            typeof window.showDetails === 'function' &&
            window.currentEvent &&
            window.site
        ) {
            window.showDetails();
            return true;
        }
        return false;
    }

    const timer = setInterval(() => {
        if (tryOpenDetails()) {
            clearInterval(timer);
        }
    }, 1000);
})();
